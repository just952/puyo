    public boolean rotateCounterClockwise() {
        int originalRotation = currentPair.getRotation();
        currentPair.rotateCounterClockwise();
        if (board.canPlace(currentPair)) {
            resetLockDelay();
            return true;
        } else {
            currentPair.setRotation(originalRotation);
            return false;
        }
    }
